public function addItemsToOrder(Request $request, string $uuid)
    {
        $auth = auth()->user();
        DB::beginTransaction();

        try {

            $order = OrderMenuRestaurant::where('uuid', $uuid)->firstOrFail();

            $validated = $request->validate([
                'menus' => ['nullable', 'array'],
                'menus.*.menus_restaurant_uuid' => ['required', 'uuid', 'exists:menus_restaurants,uuid'],
                'menus.*.quantity' => ['required', 'numeric', 'min:1'],
                'menus.*.unit_price' => ['nullable', 'numeric', 'min:1'],

                'drinks' => ['nullable', 'array'],
                'drinks.*.product_uuid' => ['required', 'uuid', 'exists:produits,uuid'],
                'drinks.*.quantity' => ['required', 'numeric', 'min:'],
                'drinks.*.unit_price' => ['nullable', 'numeric', 'min:0'],
            ]);

            $warehouseUuid = $validated['warehouse_uuid']
                ?? Warehouse::where('is_used_for_restaurant', true)->firstOrFail()->uuid;

            /*
            |--------------------------------------------------------------------------
            | Vérification stock
            |--------------------------------------------------------------------------
            */

            $menuErrors = $this->verifyMenuStock($validated['menus'] ?? [], $warehouseUuid);
            if ($menuErrors) {
                return response()->json([
                    'status' => 'error',
                    'message' => 'Stock insuffisant pour certains menus.',
                    'details' => $menuErrors,
                ], 422);
            }

            if (!empty($validated['drinks'])) {
                $drinkErrors = $this->verifyBarStock($validated['drinks']);
                if ($drinkErrors) {
                    return response()->json([
                        'status' => 'error',
                        'message' => 'Stock insuffisant pour certaines boissons.',
                        'details' => $drinkErrors,
                    ], 422);
                }
            }

            /*
            |--------------------------------------------------------------------------
            | MENUS
            |--------------------------------------------------------------------------
            */

            foreach ($validated['menus'] ?? [] as $m) {

                $menu = MenuRestaurant::findOrFail($m['menus_restaurant_uuid']);
                $unitPrice = $m['unit_price'] ?? $menu->price ?? 0;
                $isLastItem = $m['is_last_items'] ?? false;

                if ($isLastItem) {
                    continue;
                }

                $existingItem = OrderMenuRestaurantItem::where('order_menu_restaurant_uuid', $order->uuid)->where('menus_restaurant_uuid', $menu->uuid)
                    ->first();

                if ($existingItem) {

                    $newQty = $m['quantity'];
                    $oldQty = $existingItem->quantity;

                    $isRejected = in_array($existingItem->status, [
                        OrderMenuRestaurantItemStatus::REJECTED->value,
                        OrderMenuRestaurantItemStatus::NEW_REJECTED->value
                    ]);

                    $isRejectedForUpdate = $existingItem->status ===
                        OrderMenuRestaurantItemStatus::REJECTED_FOR_NEW_UPDATE->value;

                    $statusesToTransfer = in_array($existingItem->status, [
                        OrderMenuRestaurantItemStatus::DELIVERED->value,
                        OrderMenuRestaurantItemStatus::PARTIAL_DELIVERED->value,
                    ]);

                    // ⚡ évite traitement inutile
                    if ($newQty == $oldQty && !$isRejected && !$isRejectedForUpdate && !$statusesToTransfer) {
                        continue;
                    }

                    /*
                    |--------------------------------------------------------------------------
                    | 🔸 CAS : IN_PREPARATION
                    |--------------------------------------------------------------------------
                    */
                    if ($existingItem->status === OrderMenuRestaurantItemStatus::IN_PREPARATION->value) {

                        // ❌ réduire interdit
                        if ($newQty < $oldQty) {
                            return response()->json([
                                'status' => 'error',
                                'message' => "Impossible de réduire \"{$menu->name}\" en préparation. Vous ne pouvez que augmenter la quantité.",
                            ], 422);
                        }

                        // ✅ augmenter
                        $diff = $newQty - $oldQty;

                        $existingItem->update([
                            'quantity' => $newQty,
                            'quantity_exactly' => $newQty,
                            'total_price' => $unitPrice * $newQty,
                            'status' => OrderMenuRestaurantItemStatus::TRANSFERRED->value,
                            'updated_by' => $auth->id,
                        ]);

                        if ($diff > 0) {
                        $components = MenuOrderItem::where('menus_restaurant_uuid', $menu->uuid)->get();
                        foreach ($components as $comp) {
                            VirtualOrderMenuRestaurant::create([
                                'orders_menu_restaurant_uuid' => $order->uuid,
                                'item_uuid' => $existingItem->uuid,
                                'product_uuid' => $comp->product_uuid,
                                'quantity_reserved' => $diff * $comp->quantity_used,
                                'quantity_exactly' => $diff * $comp->quantity_used,
                                'quantity_delivered_exactly' => $diff * $comp->quantity_used,
                                'created_by' => $auth->id,
                                'updated_by' => $auth->id,
                                'is_new_items' => true,
                                'is_last_items' => false
                            ]);
                        }
                        }

                        continue;
                    }


                    if (in_array($existingItem->status, [
                        OrderMenuRestaurantItemStatus::DELIVERED->value,
                        OrderMenuRestaurantItemStatus::PARTIAL_DELIVERED->value,
                    ])) {
                        // calcul de la différence réelle
                        $diff = $newQty - $oldQty;

                        if ($diff < 0) {
                            return response()->json([
                                'status' => 'error',
                                'message' => "Impossible de réduire \"{$menu->name}\" déjà servie ou partiellement servie. Vous ne pouvez que augmenter la quantité.",
                            ], 422);
                        }

                        // ✅ diff > 0 → augmentation autorisée
                        if ($diff > 0) {
                            $existingItem->update([
                                'quantity' => $newQty,
                                'quantity_exactly' => $existingItem->quantity_exactly + $diff,
                                'total_price' => $unitPrice * $newQty,
                                'status' => OrderMenuRestaurantItemStatus::TRANSFERRED->value,
                                'updated_by' => $auth->id,
                            ]);

                            // 🔥 recréer les virtual items uniquement pour cet item
                            $components = MenuOrderItem::where('menus_restaurant_uuid', $menu->uuid)->get();
                            foreach ($components as $comp) {
                                VirtualOrderMenuRestaurant::create([
                                    'orders_menu_restaurant_uuid' => $order->uuid,
                                    'item_uuid' => $existingItem->uuid,
                                    'product_uuid' => $comp->product_uuid,
                                    'quantity_reserved' => $diff * $comp->quantity_used,
                                    'quantity_exactly' => $diff * $comp->quantity_used,
                                    'quantity_delivered_exactly' => $diff * $comp->quantity_used,
                                    'created_by' => $auth->id,
                                    'updated_by' => $auth->id,
                                    'is_new_items' => true,
                                    'is_last_items' => false,
                                ]);
                            }
                        }

                        // ✅ si diff = 0 → on ne touche pas à l’item
                        continue; // passe au prochain item
                    }

                    if ($isRejectedForUpdate) {
                        // ❌ réduire interdit
                        if ($newQty < $oldQty) {
                            return response()->json([
                                'status' => 'error',
                                'message' => "Impossible de réduire \"{$menu->name}\" (rejet du prêt). Vous ne pouvez qu’augmenter la quantité.",
                            ], 422);
                        }

                        // ✅ augmenter ou égal
                        $diff = $newQty - $oldQty;

                        $existingItem->update([
                            'quantity' => $newQty,
                            'quantity_exactly' => $newQty,
                            'total_price' => $unitPrice * $newQty,
                            'status' => OrderMenuRestaurantItemStatus::TRANSFERRED->value,
                            'updated_by' => $auth->id,
                        ]);

                        if ($diff > 0) {
                            $components = MenuOrderItem::where('menus_restaurant_uuid', $menu->uuid)->get();

                            foreach ($components as $comp) {
                                VirtualOrderMenuRestaurant::create([
                                    'orders_menu_restaurant_uuid' => $order->uuid,
                                    'item_uuid' => $existingItem->uuid,
                                    'product_uuid' => $comp->product_uuid,
                                    'quantity_reserved' => $diff * $comp->quantity_used,
                                    'quantity_exactly' => $diff * $comp->quantity_used,
                                    'quantity_delivered_exactly' => $diff * $comp->quantity_used,
                                    'created_by' => $auth->id,
                                    'updated_by' => $auth->id,
                                    'is_new_items' => true,
                                    'is_last_items' => false
                                ]);
                            }
                        }

                        continue;
                    }


                    /*
                    |--------------------------------------------------------------------------
                    | 🔸 CAS : REJECTED / NEW_REJECTED
                    |--------------------------------------------------------------------------
                    */
                    if ($isRejected) {

                        if ($newQty > $oldQty) {
                            return response()->json([
                                'status' => 'error',
                                'message' => "Impossible d’augmenter \"{$menu->name}\" rejeté. Vous ne pouvez que réduire la quantité.",
                            ], 422);
                        }

                        // ✅ réduire
                        $reduction = $oldQty - $newQty;

                        $existingItem->update([
                            'quantity' => $newQty,
//                            'quantity_exactly' => $existingItem->quantity_exactly + $newQty,
                            'total_price' => $unitPrice * $newQty,
                            'status' => OrderMenuRestaurantItemStatus::TRANSFERRED->value,
                            'updated_by' => $auth->id,
                        ]);

                        if ($reduction > 0) {
                            $components = MenuOrderItem::where('menus_restaurant_uuid', $menu->uuid)->get();

                            foreach ($components as $comp) {
                                VirtualOrderMenuRestaurant::create([
                                    'orders_menu_restaurant_uuid' => $order->uuid,
                                    'item_uuid' => $existingItem->uuid,
                                    'product_uuid' => $comp->product_uuid,
                                    'quantity_reserved' => -($reduction * $comp->quantity_used),
                                    'quantity_exactly' => -($reduction * $comp->quantity_used),
                                    'quantity_delivered_exactly' => -($reduction * $comp->quantity_used),
                                    'created_by' => $auth->id,
                                    'updated_by' => $auth->id,
                                    'is_new_items' => false,
                                    'is_last_items' => false
                                ]);
                            }
                        }

                        continue;
                    }

                    /*
                    |--------------------------------------------------------------------------
                    | 🔸 AUTRES STATUTS
                    |--------------------------------------------------------------------------
                    */
                    $existingItem->update([
                        'quantity' => $newQty,
                        'quantity_exactly' => $newQty,
                        'total_price' => $unitPrice * $newQty,
                        'status' => OrderMenuRestaurantItemStatus::TRANSFERRED->value,
                        'updated_by' => $auth->id,
                    ]);

                    continue;
                }

                /*
                |--------------------------------------------------------------------------
                | 🆕 CAS 2 : NOUVEL ITEM
                |--------------------------------------------------------------------------
                */
                $item = OrderMenuRestaurantItem::create([
                    'order_menu_restaurant_uuid' => $order->uuid,
                    'menus_restaurant_uuid' => $menu->uuid,
                    'quantity' => $m['quantity'],
                    'quantity_exactly' => $m['quantity'],
                    'unit_price' => $unitPrice,
                    'total_price' => $unitPrice * $m['quantity'],
                    'status' => OrderMenuRestaurantItemStatus::TRANSFERRED->value,
                    'created_by' => $auth->id,
                    'updated_by' => $auth->id,
                    'is_new_items' => true,
                    'is_last_items' => false
                ]);

                $components = MenuOrderItem::where('menus_restaurant_uuid', $menu->uuid)->get();

                foreach ($components as $comp) {
                    VirtualOrderMenuRestaurant::create([
                        'orders_menu_restaurant_uuid' => $order->uuid,
                        'item_uuid' => $item->uuid,
                        'product_uuid' => $comp->product_uuid,
                        'quantity_reserved' => $m['quantity'] * $comp->quantity_used,
                        'quantity_exactly' => $m['quantity'] * $comp->quantity_used,
                        'quantity_delivered_exactly' => $m['quantity'] * $comp->quantity_used,
                        'created_by' => $auth->id,
                        'updated_by' => $auth->id,
                        'is_new_items' => true,
                        'is_last_items' => false
                    ]);
                }
            }



            /*
            |--------------------------------------------------------------------------
            | DRINKS
            |--------------------------------------------------------------------------
            */

            foreach ($validated['drinks'] ?? [] as $d) {

                $isLastItem = $d['is_last_items'] ?? false;

                if ($isLastItem) {
                    continue;
                }

                $unitPrice = $d['unit_price'] ?? 0;
                $product = Product::findOrFail($d['product_uuid']);

                $existingDrink = OrderRestaurantDrink::where('order_menu_restaurant_uuid', $order->uuid)
                    ->where('product_uuid', $d['product_uuid'])
                    ->first();

                /*
                |--------------------------------------------------------------------------
                | 🔥 CAS 1 : DRINK EXISTE
                |--------------------------------------------------------------------------
                */
                if ($existingDrink) {

                    $newQty = $d['quantity'];
                    $oldQty = $existingDrink->quantity;

                    $isRejectedDrinks = in_array($existingDrink->status, [
                        OrderMenuRestaurantItemStatus::REJECTED->value,
                        OrderMenuRestaurantItemStatus::NEW_REJECTED->value
                    ]);

                    $isRejectedForUpdateDrinks = $existingDrink->status ===
                        OrderMenuRestaurantItemStatus::REJECTED_FOR_NEW_UPDATE->value;

                    $statusesToTransferDrinks = in_array($existingDrink->status, [
                        OrderMenuRestaurantItemStatus::DELIVERED->value,
                        OrderMenuRestaurantItemStatus::PARTIAL_DELIVERED->value,
                    ]);

                    // ⚡ éviter traitement inutile
                    if ($newQty == $oldQty && !$isRejectedDrinks && !$isRejectedForUpdateDrinks && !$statusesToTransferDrinks) {
                        continue;
                    }

                    /*
                    |--------------------------------------------------------------------------
                    | 🔸 CAS : IN_PREPARATION
                    |--------------------------------------------------------------------------
                    */
                    if ($existingDrink->status === OrderMenuRestaurantItemStatus::IN_PREPARATION->value) {

                        // ❌ réduire interdit
                        if ($newQty < $oldQty) {
                            return response()->json([
                                'status' => 'error',
                                'message' => "Impossible de réduire \"{$product->name}\" (en préparation). Vous ne pouvez que augmenter la quantité.",
                            ], 422);
                        }

                        // ✅ augmenter
                        $diff = $newQty - $oldQty;

                        $existingDrink->update([
                            'quantity' => $newQty,
                            'quantity_exactly' => $newQty,
                            'total_price' => $unitPrice * $newQty,
                            'status' => OrderMenuRestaurantItemStatus::TRANSFERRED->value,
                            'updated_by' => $auth->id,
                        ]);

                        VirtualOrderMenuRestaurant::create([
                            'orders_menu_restaurant_uuid' => $order->uuid,
                            'item_uuid' => $existingDrink->uuid,
                            'product_uuid' => $d['product_uuid'],
                            'quantity_reserved' => $diff,
                            'quantity_exactly' => $diff,
                            'quantity_delivered_exactly' => $diff,
                            'created_by' => $auth->id,
                            'updated_by' => $auth->id,
                            'is_new_items' => true,
                            'is_last_items' => false
                        ]);

                        continue;
                    }


                    if ($isRejectedForUpdateDrinks) {

                        // ❌ réduire interdit
                        if ($newQty < $oldQty) {
                            return response()->json([
                                'status' => 'error',
                                'message' => "Impossible de réduire \"{$product->name}\" (rejet du prêt). Vous ne pouvez qu’augmenter la quantité.",
                            ], 422);
                        }

                        // ✅ augmenter ou égal
                        $diff = $newQty - $oldQty;

                        $existingDrink->update([
                            'quantity' => $newQty,
                            'quantity_exactly' => $newQty,
                            'total_price' => $unitPrice * $newQty,
                            'status' => OrderMenuRestaurantItemStatus::TRANSFERRED->value,
                            'updated_by' => $auth->id,
                        ]);

                        if ($diff > 0) {
                            VirtualOrderMenuRestaurant::create([
                                'orders_menu_restaurant_uuid' => $order->uuid,
                                'item_uuid' => $existingDrink->uuid,
                                'product_uuid' => $d['product_uuid'],
                                'quantity_reserved' => $diff,
                                'quantity_exactly' => $diff,
                                'quantity_delivered_exactly' => $diff,
                                'created_by' => $auth->id,
                                'updated_by' => $auth->id,
                                'is_new_items' => true,
                                'is_last_items' => false
                            ]);
                        }

                        continue;
                    }



                    if (in_array($existingDrink->status, [
                        OrderMenuRestaurantItemStatus::DELIVERED->value,
                        OrderMenuRestaurantItemStatus::PARTIAL_DELIVERED->value,
                    ])) {
                        // calculer la différence réelle
                        $diff = $newQty - $oldQty;

                        // ❌ réduire interdit
                        if ($diff < 0) {
                            return response()->json([
                                'status' => 'error',
                                'message' => "Impossible de réduire \"{$product->name}\" déjà livré ou partiellement livré. Vous ne pouvez qu’augmenter la quantité.",
                            ], 422);
                        }

                        // ✅ si diff > 0 → seulement alors on repasse à TRANSFERRED
                        if ($diff > 0) {
                            $existingDrink->update([
                                'quantity' => $newQty,
                                'quantity_exactly' => $existingDrink->quantity_exactly + $diff, // accumuler correctement
                                'total_price' => $unitPrice * $newQty,
                                'status' => OrderMenuRestaurantItemStatus::TRANSFERRED->value,
                                'updated_by' => $auth->id,
                            ]);

                            // 🔥 recréer les virtual items seulement si on a augmenté
                            VirtualOrderMenuRestaurant::create([
                                'orders_menu_restaurant_uuid' => $order->uuid,
                                'item_uuid' => $existingDrink->uuid,
                                'product_uuid' => $d['product_uuid'],
                                'quantity_reserved' => $diff,
                                'quantity_exactly' => $diff,
                                'quantity_delivered_exactly' => $diff,
                                'created_by' => $auth->id,
                                'updated_by' => $auth->id,
                                'is_new_items' => true,
                                'is_last_items' => false,
                            ]);
                        }

                        continue; // passe au prochain drink
                    }


                    /*
                    |--------------------------------------------------------------------------
                    | 🔸 CAS : REJECTED / NEW_REJECTED
                    |--------------------------------------------------------------------------
                    */
                    if ($isRejectedDrinks) {

                        // ❌ augmenter interdit
                        if ($newQty > $oldQty) {
                            return response()->json([
                                'status' => 'error',
                                'message' => "Impossible d’augmenter \"{$product->name}\" (rejeté). Vous ne pouvez que réduire la quantité.",
                            ], 422);
                        }

                        // ✅ réduire
                        $diff = $oldQty - $newQty;

                        $existingDrink->update([
                            'quantity' => $newQty,
                            'quantity_exactly' => $newQty,
                            'total_price' => $unitPrice * $newQty,
                            'status' => OrderMenuRestaurantItemStatus::TRANSFERRED->value,
                            'updated_by' => $auth->id,
                        ]);

                        if ($diff > 0) {
                            VirtualOrderMenuRestaurant::create([
                                'orders_menu_restaurant_uuid' => $order->uuid,
                                'item_uuid' => $existingDrink->uuid,
                                'product_uuid' => $d['product_uuid'],
                                'quantity_reserved' => -$diff,
                                'quantity_exactly' => -$diff,
                                'quantity_delivered_exactly' => -$diff,
                                'created_by' => $auth->id,
                                'updated_by' => $auth->id,
                                'is_new_items' => false,
                                'is_last_items' => false
                            ]);
                        }

                        continue;
                    }


                    /*
                    |--------------------------------------------------------------------------
                    | 🔸 AUTRES STATUTS
                    |--------------------------------------------------------------------------
                    */
                    $existingDrink->update([
                        'quantity' => $newQty,
                        'quantity_exactly' => $newQty,
                        'total_price' => $unitPrice * $newQty,
                        'status' => OrderMenuRestaurantItemStatus::TRANSFERRED->value,
                        'updated_by' => $auth->id,
                    ]);

                    continue;
                }
                /*
                |--------------------------------------------------------------------------
                | 🆕 CAS 2 : NOUVEAU DRINK
                |--------------------------------------------------------------------------
                */
                $drink = OrderRestaurantDrink::create([
                    'order_menu_restaurant_uuid' => $order->uuid,
                    'product_uuid' => $d['product_uuid'],
                    'quantity' => $d['quantity'],
                    'quantity_exactly' => $d['quantity'],
                    'unit_price' => $unitPrice,
                    'total_price' => $unitPrice * $d['quantity'],
                    'status' => OrderMenuRestaurantItemStatus::TRANSFERRED->value,
                    'created_by' => $auth->id,
                    'updated_by' => $auth->id,
                    'is_new_items' => true,
                    'is_last_items' => false
                ]);

                VirtualOrderMenuRestaurant::create([
                    'orders_menu_restaurant_uuid' => $order->uuid,
                    'item_uuid' => $drink->uuid,
                    'product_uuid' => $d['product_uuid'],
                    'quantity_reserved' => $d['quantity'],
                    'quantity_exactly' => $d['quantity'],
                    'quantity_delivered_exactly' => $d['quantity'],
                    'created_by' => $auth->id,
                    'updated_by' => $auth->id,
                    'is_new_items' => true,
                    'is_last_items' => false
                ]);
            }

            $this->refreshOrderStatus($order);

            // 🔹 Update order
            $order->update([
                'updated_by' => $auth->id,
            ]);

            DB::commit();

            return response()->json([
                'status' => 'success',
                'message' => 'Commande mise à jour avec succès'
            ]);

        } catch (\Throwable $e) {

            DB::rollBack();

            // 🔹 Loguer l'erreur complète dans le fichier de logs Laravel
            Log::error('Erreur lors de l’ajout des éléments à la commande', [
                'order_uuid' => $uuid,
                'user_id' => $auth->id ?? null,
                'error_message' => $e->getMessage(),
                'trace' => $e->getTraceAsString(),
            ]);

            // 🔹 Retourner le message exact au frontend (optionnel, attention aux données sensibles)
            return response()->json([
                'status' => 'error',
                'message' => 'Erreur lors de l’ajout des éléments : ' . $e->getMessage(),
            ], 500);
        }
    }
